-- Helpers ----------------------------------------------------
local function safeSetClipboard(text)
	local ok = pcall(setclipboard, text)
	if not ok then pcall(function() clipboard.set(text) end) end
end

local function readClipboard()
	local ok, v = pcall(function() return getclipboard() end)
	if ok and type(v) == "string" then return v end
	ok, v = pcall(function() return readclipboard() end)
	if ok and type(v) == "string" then return v end
	ok, v = pcall(function() return clipboard.get() end)
	if ok and type(v) == "string" then return v end
	return nil
end

local function httpGet(url)
	local ok, res = pcall(function() return game:HttpGet(url, true) end)
	if ok and type(res) == "string" then return res end
	ok, res = pcall(function() return HttpGet(url) end)
	if ok and type(res) == "string" then return res end
	return nil
end

local function decodeJSON(str)
	local ok, data = pcall(function() return HttpService:JSONDecode(str) end)
	if ok and type(data) == "table" then return data end
	return nil
end

-- Game name: several sources in order, because GetProductInfo gets
-- rate limited quickly and some executors block it entirely
local function fetchNameViaApi(placeId)
	local uniStr = httpGet("https://apis.roblox.com/universes/v1/places/" .. tostring(placeId) .. "/universe")
	local uni = decodeJSON(uniStr)
	local uid = type(uni) == "table" and uni.universeId
	if not uid then return nil end
	local gStr = httpGet("https://games.roblox.com/v1/games?universeIds=" .. tostring(uid))
	local g = decodeJSON(gStr)
	if type(g) == "table" and type(g.data) == "table" and type(g.data[1]) == "table" then
		local n = g.data[1].Name
		if type(n) == "string" and n ~= "" then return n end
	end
	return nil
end

local function getPlaceName(placeId)
	-- 1. MarketplaceService (the proper display name)
	for attempt = 1, 2 do
		local ok, info = pcall(function()
			return MarketplaceService:GetProductInfo(placeId, Enum.AssetDependencyType.Place)
		end)
		if ok and type(info) == "table" and type(info.Name) == "string" and info.Name ~= "" then
			return info.Name
		end
		if attempt == 1 then task.wait(0.5) end
	end
	-- 2. Roblox games API by universe id (used when 1. is throttled)
	local okApi, name = pcall(fetchNameViaApi, placeId)
	if okApi and name then return name end
	-- 3. DataModel name (always available locally)
	if type(game.Name) == "string" and game.Name ~= "" then
		return game.Name
	end
	return nil
end

local function getExecutorName()
	-- different executors expose different names for this API
	local ok, name = pcall(function() return identifyexecutor() end)
	if not ok or type(name) ~= "string" or name == "" then
		ok, name = pcall(function() return getexecutorname() end)
	end
	if ok and type(name) == "string" and name ~= "" then return name end
	return "Unknown"
end

local function maskValue(value)
	local s = tostring(value)
	return string.rep("*", math.min(math.max(#s, 6), 12))
end

-- Max players ------------------------------------------------
-- DataModel has NO MaxPlayers member: reading game.MaxPlayers throws
-- "not a valid member" and kills the whole script (that one line was
-- silently disabling verify, the executor label and the link buttons).
-- The value is read defensively here and filled from the games API.
local cachedMaxPlayers = nil
local maxPlayersRow = nil -- forward ref, assigned when the card row exists

local function maxPlayersText()
	if cachedMaxPlayers then return tostring(cachedMaxPlayers) end
	-- try every place a runtime might expose it (all guarded)
	local probes = {
		function() return game.MaxPlayers end,
		function() return game.Players.MaxPlayers end,
		function() return game:GetAttribute("MaxPlayers") end,
	}
	for _, probe in ipairs(probes) do
		local ok, v = pcall(probe)
		if ok and type(v) == "number" and v > 0 then
			cachedMaxPlayers = v
			return tostring(v)
		end
	end
	return "Loading..."
end

local function applyMaxPlayers()
	if not cachedMaxPlayers or not maxPlayersRow then return end
	maxPlayersRow.value = tostring(cachedMaxPlayers)
	local lbl = maxPlayersRow.labelRef
	if lbl and lbl.Parent then lbl.Text = maxPlayersRow.value end
end

-- GUI builders -----------------------------------------------
local function New(class, props)
	local inst = Instance.new(class)
	for k, v in pairs(props) do
		inst[k] = v
	end
	return inst
end

local function corner(inst, radius)
	return New("UICorner", {CornerRadius = UDim.new(0, radius), Parent = inst})
end

local function uiStroke(inst, color, thickness, transparency)
	return New("UIStroke", {
		Color = color,
		Thickness = thickness,
		Transparency = transparency or 0,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
		Parent = inst,
	})
end

local function uiGradient(inst, keypoints, rotation, transparencySeq)
	local g = New("UIGradient", {
		Color = ColorSequence.new(keypoints),
		Rotation = rotation or 0,
		Parent = inst,
	})
	if transparencySeq then g.Transparency = transparencySeq end
	return g
end

local function label(parent, props)
	local base = {
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		Font = Enum.Font.GothamSemibold,
		Text = "",
		TextColor3 = C.text,
		TextSize = 12,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextYAlignment = Enum.TextYAlignment.Center,
	}
	for k, v in pairs(props) do base[k] = v end
	base.Parent = parent
	return New("TextLabel", base)
end

-- Icon chip (light plate + tinted glyph, readable on any plate)
-- Interactive chip colour state ------------------------------------
-- Every chip records its BASE colour here. Hover effects and the green
-- "copied" flash animate back to that base (or to the hover colour while
-- the mouse is still over the chip). They never restore a colour that was
-- captured mid-hover - that is exactly what left buttons stuck white.
local chipState = {}

local function applyChipColors(chip, instant)
	local s = chipState[chip]
	if not s or not chip.Parent then return end
	local bg, tr
	if s.hovered and s.hover then
		bg, tr = s.hover, s.hoverTrans
	else
		bg, tr = s.base, s.baseTrans
	end
	if instant then
		chip.BackgroundColor3 = bg
		chip.BackgroundTransparency = tr
	else
		TweenService:Create(chip, TweenInfo.new(0.14), {
			BackgroundColor3 = bg, BackgroundTransparency = tr,
		}):Play()
	end
end

local function iconChip(parent, x, y, size, iconId, opts)
	opts = opts or {}
	local z = opts.z or 16
	local isButton = opts.button == true
	local tone = opts.tone and TONE[opts.tone]
	local baseBg = opts.bg or (tone and tone.plate) or C.chip
	local baseTrans = opts.bgTrans or 0.05
	local props = {
		Position = UDim2.new(0, x, 0, y),
		Size = UDim2.new(0, size, 0, size),
		BackgroundColor3 = baseBg,
		BackgroundTransparency = baseTrans,
		BorderSizePixel = 0,
		ZIndex = z,
		Parent = parent,
	}
	if isButton then
		props.Text = ""
		props.AutoButtonColor = false
	end
	local chip = New(isButton and "TextButton" or "Frame", props)
	-- register the base colour so hover/flash effects can always return to it
	chipState[chip] = {base = baseBg, baseTrans = baseTrans}
	corner(chip, opts.radius or math.max(6, math.floor(size * 0.3)))
	if opts.stroke ~= false then
		uiStroke(chip, opts.strokeColor or Color3.fromRGB(255, 255, 255), 1, opts.strokeTrans or 0.8)
	end
	local iconSize = math.max(10, math.floor(size * (opts.iconScale or 0.58)))
	local img = New("ImageLabel", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.new(0.5, 0, 0.5, 0),
		Size = UDim2.new(0, iconSize, 0, iconSize),
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		Image = iconId or "",
		ImageColor3 = opts.tint or (tone and tone.glyph) or C.icon,
		ScaleType = Enum.ScaleType.Fit,
		ZIndex = z + 1,
		Parent = chip,
	})
	return chip, img
end

local function iconButton(parent, x, y, size, iconId, onClick, opts)
	opts = opts or {}
	local chip, img = iconChip(parent, x, y, size, iconId, {
		button = true, z = opts.z, radius = opts.radius,
		bg = opts.bg, bgTrans = opts.bgTrans, iconScale = opts.iconScale,
		strokeColor = opts.strokeColor, strokeTrans = opts.strokeTrans,
		tint = opts.tint, tone = opts.tone,
	})
	-- Hover: a gentle lift of the chip's OWN colour (never a solid white
	-- blob - light plates looked wrong on the dark UI). State lives in
	-- chipState, so the leave/flash handlers always restore the base.
	local s = chipState[chip]
	if s then
		s.hover = opts.hoverBg or s.base:Lerp(Color3.new(1, 1, 1), 0.32)
		s.hoverTrans = opts.hoverTrans or math.max(0, s.baseTrans * 0.4)
	end
	chip.MouseEnter:Connect(function()
		if s then s.hovered = true end
		applyChipColors(chip)
	end)
	chip.MouseLeave:Connect(function()
		if s then s.hovered = false end
		applyChipColors(chip)
		-- restore size in case the press ended outside the chip
		TweenService:Create(chip, TweenInfo.new(0.14, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
			Size = UDim2.new(0, size, 0, size)
		}):Play()
	end)
	-- press feedback: quick squeeze + flash
	chip.MouseButton1Down:Connect(function()
		TweenService:Create(chip, TweenInfo.new(0.05), {
			Size = UDim2.new(0, size - 3, 0, size - 3),
		}):Play()
	end)
	chip.MouseButton1Up:Connect(function()
		TweenService:Create(chip, TweenInfo.new(0.14, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
			Size = UDim2.new(0, size, 0, size),
		}):Play()
	end)
	if onClick then
		chip.MouseButton1Click:Connect(function()
			onClick(chip, img)
		end)
	end
	return chip, img
end

local function flashCopied(img, chip)
	if not img then return end
	local prevIcon = img.Image
	-- only used as a fallback for chips with no registered base colour
	local prevBg = chip and chipState[chip] == nil and chip.BackgroundColor3 or nil
	img.Image = ICON.check
	if chip then
		chip.BackgroundColor3 = Color3.fromRGB(150, 255, 195)
	end
	task.delay(1.2, function()
		if img.Parent then img.Image = prevIcon end
		if chip and chip.Parent then
			if chipState[chip] then
				-- back to the registered base colour (or the hover colour
				-- if the mouse is still on the chip) - never to a colour
				-- captured while hovered, which could be white
				applyChipColors(chip)
			elseif prevBg then
				chip.BackgroundColor3 = prevBg
			end
		end
	end)
end

-- Toast notification (slides up from the bottom of a window) ----
local liveToasts = {}

local TOAST_STYLE = {
	success = {bg = Color3.fromRGB(14, 74, 46), stroke = C.green,  icon = ICON.check,       iconBg = Color3.fromRGB(10, 62, 38),  glyph = Color3.fromRGB(150, 255, 205)},
	error   = {bg = Color3.fromRGB(104, 16, 34), stroke = C.red,   icon = ICON.alertCircle, iconBg = Color3.fromRGB(86, 12, 28),   glyph = Color3.fromRGB(255, 170, 185)},
	info    = {bg = Color3.fromRGB(30, 40, 78),  stroke = Color3.fromRGB(120, 160, 255), icon = ICON.bell, iconBg = Color3.fromRGB(24, 34, 66), glyph = Color3.fromRGB(170, 198, 255)},
}

local function showToast(parent, text, kind)
	if not parent or not parent.Parent then return end
	local st = TOAST_STYLE[kind or "success"] or TOAST_STYLE.success

	local toast = New("Frame", {
		AnchorPoint = Vector2.new(0.5, 1),
		Position = UDim2.new(0.5, 0, 1, 20),
		Size = UDim2.new(0, 330, 0, 44),
		BackgroundColor3 = st.bg,
		BackgroundTransparency = 0.05,
		BorderSizePixel = 0,
		ZIndex = 56,
		Parent = parent,
	})
	corner(toast, 22)
	uiStroke(toast, st.stroke, 1.5, 0.35)
	uiGradient(toast, {
		ColorSequenceKeypoint.new(0, Color3.new(1, 1, 1)),
		ColorSequenceKeypoint.new(1, Color3.fromRGB(0.75, 0.75, 0.75)),
	}, 90)

	iconChip(toast, 8, 8, 28, st.icon, {z = 58, radius = 14, bg = st.iconBg, bgTrans = 0.06, tint = st.glyph})
	local txt = label(toast, {
		Position = UDim2.new(0, 46, 0, 0),
		Size = UDim2.new(1, -56, 1, 0),
		Text = text,
		Font = Enum.Font.GothamBold,
		TextSize = 12,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextTruncate = Enum.TextTruncate.AtEnd,
		ZIndex = 58,
	})

	-- slide in
	TweenService:Create(toast, TweenInfo.new(0.34, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
		Position = UDim2.new(0.5, 0, 1, -14),
	}):Play()

	-- push older toasts up so they stack
	for _, other in ipairs(liveToasts) do
		if other.Parent then
			local p = other.Position
			TweenService:Create(other, TweenInfo.new(0.28, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
				Position = UDim2.new(p.X.Scale, p.X.Offset, p.Y.Scale, p.Y.Offset - 52),
			}):Play()
		end
	end
	table.insert(liveToasts, toast)

	task.delay(2.4, function()
		if not toast.Parent then return end
		local out = TweenService:Create(toast, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
			Position = UDim2.new(0.5, 0, 1, 20),
			BackgroundTransparency = 1,
		})
		out:Play()
		for _, d in ipairs(toast:GetDescendants()) do
			if d:IsA("TextLabel") then
				TweenService:Create(d, TweenInfo.new(0.22), {TextTransparency = 1}):Play()
			elseif d:IsA("ImageLabel") then
				TweenService:Create(d, TweenInfo.new(0.22), {ImageTransparency = 1}):Play()
			elseif d:IsA("UIStroke") then
				TweenService:Create(d, TweenInfo.new(0.22), {Transparency = 1}):Play()
			end
		end
		out.Completed:Wait()
		local idx = table.find(liveToasts, toast)
		if idx then table.remove(liveToasts, idx) end
		toast:Destroy()
	end)

	return toast
end

-- Link buttons (copy URL to the clipboard + toast) -------------
-- Defined here and wired the moment each button is created, so no
-- problem anywhere in the loader section can disable them
local function linkButton(btn, link, parent)
	btn.MouseButton1Click:Connect(function()
		safeSetClipboard(link)
		-- makeButton returns a TextButton chip, so search for its glyph
		local img = btn:FindFirstChildWhichIsA("ImageLabel", true)
		if img then flashCopied(img, img.Parent) end
		showToast(parent, "Link copied  —  " .. link, "success")
	end)
end

-- Icon + text button -----------------------------------------
local function makeButton(parent, opts)
	local z = opts.z or 22
	local primary = opts.kind ~= "ghost"
	local btn = New("TextButton", {
		Position = UDim2.new(0, opts.x, 0, opts.y),
		Size = UDim2.new(0, opts.w, 0, opts.h),
		BackgroundColor3 = primary and Color3.fromRGB(215, 20, 45) or Color3.fromRGB(38, 5, 12),
		BorderSizePixel = 0,
		AutoButtonColor = false,
		Text = "",
		ZIndex = z,
		Parent = parent,
	})
	corner(btn, opts.radius or 12)
	local st = uiStroke(btn, primary and Color3.fromRGB(255, 140, 160) or Color3.fromRGB(150, 30, 55), 1.4, 0.3)
	uiGradient(btn, {
		ColorSequenceKeypoint.new(0, primary and Color3.fromRGB(255, 70, 95) or Color3.fromRGB(74, 12, 26)),
		ColorSequenceKeypoint.new(1, primary and Color3.fromRGB(150, 5, 22) or Color3.fromRGB(24, 2, 6)),
	}, 90)
	local chipSize = math.min(opts.h - 14, opts.iconSize or 30)
	-- plate colour follows the button: primary buttons get a deep shade
	-- of their own red, ghost buttons get the dark plate of their tone
	-- (never a light sticker on the dark UI)
	local tone = TONE[opts.tone] or TONE.neutral
	local chipPlate, chipTint
	if primary then
		chipPlate = Color3.fromRGB(132, 10, 34)
		chipTint = Color3.fromRGB(255, 255, 255)
	else
		chipPlate = tone.plate
		chipTint = tone.glyph
	end
	local chip = iconChip(btn, 12, math.floor((opts.h - chipSize) / 2), chipSize, opts.icon, {
		z = z + 1, bg = chipPlate, bgTrans = 0.1, tint = chipTint,
	})

	-- looping shine sweep that travels across the button face
	-- (kept strictly inside the button bounds so it never spills out)
	local shine = New("Frame", {
		Position = UDim2.new(0, 0, 0, 4),
		Size = UDim2.new(0, 30, 1, -8),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BackgroundTransparency = 0.88,
		BorderSizePixel = 0,
		Rotation = 14,
		ZIndex = z + 1,
		Parent = btn,
	})
	corner(shine, 8)
	TweenService:Create(shine, TweenInfo.new(2.6, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut, -1, false, 0.7), {
		Position = UDim2.new(1, -30, 0, 4),
	}):Play()
	local gap = 12 + chipSize + 8
	local lbl = label(btn, {
		Position = UDim2.new(0, gap, 0, 0),
		Size = UDim2.new(1, -(gap + 12), 1, 0),
		Text = opts.text,
		Font = Enum.Font.GothamBold,
		TextSize = opts.textSize or 14,
		TextXAlignment = Enum.TextXAlignment.Center,
		ZIndex = z + 2,
	})
	-- press feedback: white flash + slight squeeze
	local press = New("Frame", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundColor3 = Color3.new(1, 1, 1),
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		ZIndex = z + 3,
		Parent = btn,
	})
	corner(press, opts.radius or 12)
	local baseSize = UDim2.new(0, opts.w, 0, opts.h)

	btn.MouseEnter:Connect(function()
		TweenService:Create(btn, TweenInfo.new(0.15, Enum.EasingStyle.Quad), {
			BackgroundColor3 = primary and Color3.fromRGB(255, 65, 92) or Color3.fromRGB(58, 8, 18)
		}):Play()
		TweenService:Create(st, TweenInfo.new(0.15), {
			Color = Color3.fromRGB(255, 225, 230), Transparency = 0
		}):Play()
	end)
	btn.MouseLeave:Connect(function()
		TweenService:Create(btn, TweenInfo.new(0.15, Enum.EasingStyle.Quad), {
			BackgroundColor3 = primary and Color3.fromRGB(215, 20, 45) or Color3.fromRGB(38, 5, 12)
		}):Play()
		TweenService:Create(st, TweenInfo.new(0.15), {
			Color = primary and Color3.fromRGB(255, 140, 160) or Color3.fromRGB(150, 30, 55),
			Transparency = 0.3
		}):Play()
		TweenService:Create(press, TweenInfo.new(0.12), {BackgroundTransparency = 1}):Play()
		-- restore size in case the press ended outside the button
		TweenService:Create(btn, TweenInfo.new(0.16, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Size = baseSize}):Play()

-- Fade a subtree in from hidden (staggered entrances) ----------
local function fadeGroup(root, delayTime, duration)
	duration = duration or 0.4
	local saved = {}

	local function visit(d)
		if d:IsA("TextLabel") or d:IsA("TextButton") or d:IsA("TextBox") then
			saved[#saved + 1] = {inst = d, kind = "text", a = d.TextTransparency, b = d.TextStrokeTransparency}
			d.TextTransparency = 1
			d.TextStrokeTransparency = 1
		elseif d:IsA("ImageLabel") then
			saved[#saved + 1] = {inst = d, kind = "image", a = d.ImageTransparency}
			d.ImageTransparency = 1
		elseif d:IsA("Frame") and d.BackgroundTransparency < 1 then
			saved[#saved + 1] = {inst = d, kind = "bg", a = d.BackgroundTransparency}
			d.BackgroundTransparency = 1
		elseif d:IsA("UIStroke") then
			saved[#saved + 1] = {inst = d, kind = "stroke", a = d.Transparency}
			d.Transparency = 1
		end
		for _, c in ipairs(d:GetChildren()) do
			visit(c)
		end
	end
	-- hidden in one pass, restored later; if anything fails while hiding,
	-- every value captured so far is put back immediately so no element
	-- can be left invisible forever
	if not pcall(visit, root) then
		for _, it in ipairs(saved) do
			local inst = it.inst
			if inst.Parent then
				if it.kind == "text" then
					inst.TextTransparency = it.a
					inst.TextStrokeTransparency = it.b
				elseif it.kind == "image" then
					inst.ImageTransparency = it.a
				elseif it.kind == "bg" then
					inst.BackgroundTransparency = it.a
				else
					inst.Transparency = it.a
				end
			end
		end
		warn("[Syniox] fadeGroup rolled back after an error")
		return
	end

	task.delay(delayTime or 0, function()
		local tw = TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
		for _, it in ipairs(saved) do
			local inst = it.inst
			if inst.Parent then
				if it.kind == "text" then
					TweenService:Create(inst, tw, {TextTransparency = it.a, TextStrokeTransparency = it.b}):Play()
				elseif it.kind == "image" then
					TweenService:Create(inst, tw, {ImageTransparency = it.a}):Play()
				elseif it.kind == "bg" then
					TweenService:Create(inst, tw, {BackgroundTransparency = it.a}):Play()
				else
					TweenService:Create(inst, tw, {Transparency = it.a}):Play()
				end
			end
		end
	end)
end

		TextSize = 11,

		TextTruncate = Enum.TextTruncate.AtEnd,

	local entry = {labelRef = val, value = tostring(value), hidden = false}

	-- row width is scale-based (Size = UDim2.new(1, -20, ...)),
	-- so compute it explicitly instead of reading Size.X.Offset (which is 0)
	local rowW = profW - 20

	iconButton(rf, rowW - 56, 4, 24, ICON.copy, function(chip, img)
		safeSetClipboard(entry.value)
		flashCopied(img, chip)
		showToast(KeyFrame, "Copied  —  " .. entry.value, "success")
	end, {z = 23, radius = 7, iconScale = 0.6, tone = "green"})

	local eyeChip, eyeImg = iconButton(rf, rowW - 28, 4, 24, ICON.eye, function(chip, img)
		entry.hidden = not entry.hidden
		val.Text = entry.hidden and maskValue(entry.value) or entry.value
		img.Image = entry.hidden and ICON.eyeOff or ICON.eye
		showToast(KeyFrame, entry.hidden and "Value hidden" or "Value revealed", "info")

		eyeImg.Image = ICON.eyeOff
	end

	table.insert(rows, entry)
	table.insert(rowFrames, rf)
	return rf
end

local scriptButtons = {}
local runningScript = nil

local function executeScript(data)
	if runningScript then return end
	runningScript = data
	local prevSub = LdrSubText
	setLdrStatus("Running: " .. data.name .. " ...")
	local ok, err = pcall(function()
		local src = httpGet(data.url)
		if not src or src == "" then error("Script could not be downloaded") end
		local fn = loadstring(src)
		if not fn then error("loadstring is not supported") end
		fn()
	end)
	if ok then
		setLdrStatus("Ready - " .. data.name .. " executed")
	else
		setLdrStatus("Error - script could not be executed")
		warn("[Syniox] " .. tostring(err))
		task.delay(2.6, function()
			if runningScript == nil then setLdrStatus(prevSub) end
		end)
	end
	runningScript = nil
	return ok
end

-- game name: fetched by PlaceId and refreshed periodically, because
-- live games rename themselves during events
local currentGameName = nil
local function applyGameName()
	local name = getPlaceName(game.PlaceId)
	if not name or name == "" then name = "Unknown game" end
	if name ~= currentGameName then

				if GameName.Parent then
					TweenService:Create(GameName, TweenInfo.new(0.5), {TextColor3 = C.text}):Play()
				end
			end)
		end
		if runningScript == nil then
			setLdrStatus(name .. "  •  " .. #scripts .. " script")
		end
	end
end

-- max players: DataModel exposes no MaxPlayers member, so the number
-- comes from the public games API (universe id -> games endpoint) and
-- updates the game-card row once it arrives
task.spawn(function()
	local fetched = false
	for attempt = 1, 3 do
		local uni = decodeJSON(httpGet("https://apis.roblox.com/universes/v1/places/"
			.. tostring(game.PlaceId) .. "/universe"))
		local uid = type(uni) == "table" and uni.universeId
		if uid then
			local g = decodeJSON(httpGet("https://games.roblox.com/v1/games?universeIds=" .. tostring(uid)))
			local info = type(g) == "table" and type(g.data) == "table" and g.data[1]
			if type(info) == "table" and type(info.MaxPlayers) == "number" and info.MaxPlayers > 0 then
				cachedMaxPlayers = info.MaxPlayers
				applyMaxPlayers()
				fetched = true
				break
			end
		end
		task.wait(2)
	end
	if not fetched then
		-- never leave a fake "Loading..." sitting in the row
		pcall(function()
			if maxPlayersRow then
				maxPlayersRow.value = "?"
				local lbl = maxPlayersRow.labelRef
				if lbl and lbl.Parent then lbl.Text = "?" end
			end
		end)
	end
end)

-- single place icon
local function fetchPlaceIcon(placeId)
	local url = "https://thumbnails.roblox.com/v1/places/gameicons?placeIds=" .. tostring(placeId)
		.. "&returnPolicy=PlaceHolder&size=512x512&format=Png&isCircular=false"
	local data = decodeJSON(httpGet(url))
	if type(data) == "table" and type(data.data) == "table" and data.data[1] then
		return data.data[1].imageUrl
	end
	return nil
end

local function playOpenAnim()
	-- 1) show the loader and pull it on-screen FIRST: if any decorative
	--    step fails afterwards the player still lands on a real window
	--    instead of a blank screen (the slide-in runs unguarded at the end)
	LoaderFrame.Visible = true
	LoaderFrame.Position = UDim2.new(0.5, 0, -0.7, 0)
	LoaderFrame.BackgroundTransparency = 0.4

	-- both cards drop down first, then slide back up into place
	local listBase, gameBase = ListCard.Position, GameCard.Position
	ListCard.Position = listBase + UDim2.new(0, 0, 0, 54)
	GameCard.Position = gameBase + UDim2.new(0, 0, 0, 54)

	-- 2) staggered entrances (guarded): script rows, then game card rows,
	--    then the loose text layers. A failure mid-way rolls back instead
	--    of leaving half the loader invisible.
	local restored = {}
	local okFade = pcall(function()
		local li = 0
		for _, e in ipairs(scriptButtons) do
			if e.frame.Visible then
				li = li + 1
				fadeGroup(e.frame, 0.34 + li * 0.045, 0.4)
			end
		end
		for gi, rf in ipairs(gameRowFrames) do
			fadeGroup(rf, 0.5 + gi * 0.07, 0.42)
		end

		for _, d in ipairs(LoaderFrame:GetDescendants()) do
			if d:IsA("TextLabel") or d:IsA("TextButton") or d:IsA("TextBox") then
				local inScriptList = d:IsDescendantOf(ScriptList)
				local inGameRow = d.Parent and d.Parent.Name == "GameRow"
				if not inScriptList and not inGameRow then
					table.insert(restored, {inst = d, tr = d.TextTransparency, sr = d.TextStrokeTransparency})
					d.TextTransparency = 1
					d.TextStrokeTransparency = 1
				end
			end
		end

		task.delay(0.1, function()
			for _, r in ipairs(restored) do
				if r.inst.Parent then
					TweenService:Create(r.inst, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
						TextTransparency = r.tr,
						TextStrokeTransparency = r.sr,
					}):Play()
				end
			end
		end)
	end)
	if not okFade then
		-- roll back every text layer hidden before the failure
		for _, r in ipairs(restored) do
			if r.inst.Parent then
				r.inst.TextTransparency = r.tr
				r.inst.TextStrokeTransparency = r.sr
			end
		end
		warn("[Syniox] loader entrance fell back to plain state")
	end

	-- 3) the window slide-in itself: always runs, never guarded away
	TweenService:Create(LoaderFrame, TweenInfo.new(0.55, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
		Position = UDim2.new(0.5, 0, 0.5, 0),
		BackgroundTransparency = 0.1,
	}):Play()

	task.delay(0.16, function()
		if ListCard.Parent then
			TweenService:Create(ListCard, TweenInfo.new(0.55, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
				Position = listBase,
			}):Play()
		end
	end)
	task.delay(0.26, function()
		if GameCard.Parent then
			TweenService:Create(GameCard, TweenInfo.new(0.55, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
				Position = gameBase,
			}):Play()
		end
	end)
end

local function shakeInput()
	local base = inputFrame.Position
	local seq = {10, -10, 7, -7, 4, -4, 0}
	for i, dx in ipairs(seq) do
		task.delay((i - 1) * 0.045, function()
			if inputFrame.Parent then
				inputFrame.Position = base + UDim2.new(0, dx, 0, 0)
			end
		end)
	end
end

local verifying = false

-- Bring the loader on screen. The key window only hides AFTER this ran,
-- so no animation failure can ever end on a blank screen.
local function enterLoader()
	pcall(function()
		LoaderFrame.Visible = true
		LoaderFrame.Position = UDim2.new(0.5, 0, -0.7, 0)
		LoaderFrame.BackgroundTransparency = 0.4
	end)
	playOpenAnim()
end

local function doVerify()
	if verifying then return end

	-- trim leading/trailing whitespace properly (the old anchored pattern
	-- required whitespace on BOTH sides, so a pasted key with a trailing
	-- newline never matched)
	local text = tostring(KeyBox.Text or "")
	text = text:match("^%s*(.-)%s*$") or text
	-- drop quotes/brackets that come along when the key is copied
	-- from a website (they made a correct key fail verification)
	text = text:gsub("^[%s'\"`%[%]]+", ""):gsub("[%s'\"`%[%]]+$", "")

	if text == "" then
		pcall(setStatus, "idle")
		pcall(shakeInput)
		pcall(showToast, KeyFrame, "Key field is empty", "error")
		return
	end

	verifying = true
	pcall(setStatus, "load")

	task.delay(0.45, function()
		-- pcall so a failure can never lock the "verifying" flag
		local okRun, err = pcall(function()
			-- forgiving comparison: typing the key in a different case
			-- must still verify
			if string.lower(text) == string.lower(CORRECT_KEY) then
				pcall(setStatus, "ok")
				-- loader first, key window second (see enterLoader)
				enterLoader()
				KeyFrame.Visible = false
				-- toast after the loader finishes sliding in (it starts off-screen)
				task.delay(0.65, function()
					pcall(showToast, LoaderFrame, "Verification successful - welcome " .. tostring(Player.DisplayName), "success")
				end)
			else
				pcall(setStatus, "bad")
				pcall(shakeInput)
				pcall(showToast, KeyFrame, "Invalid key, please try again", "error")
			end
		end)
		verifying = false
		if not okRun then
			warn("[Syniox] verify error: " .. tostring(err))
			-- last resort for a correct key: force a plain, fully visible
			-- loader so the player is never stuck behind the key window
			if string.lower(text) == string.lower(CORRECT_KEY) then
				pcall(function()
					LoaderFrame.Visible = true
					LoaderFrame.Position = UDim2.new(0.5, 0, 0.5, 0)
					LoaderFrame.BackgroundTransparency = 0.1
					for _, d in ipairs(LoaderFrame:GetDescendants()) do
						if (d:IsA("TextLabel") or d:IsA("TextButton") or d:IsA("TextBox"))
							and d.TextTransparency > 0 then
							d.TextTransparency = 0
						end
					end
					KeyFrame.Visible = false
				end)
			end
		end
	end)
end

verifyBtn.MouseButton1Click:Connect(doVerify)
KeyBox.FocusLost:Connect(function(enterPressed)
	if enterPressed then doVerify() end
end)
